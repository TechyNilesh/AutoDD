\section{Theoretical Proofs}\label{appendix:proofs}

This section provides detailed proofs for the theoretical results presented in the main paper.

\subsection{Preliminaries and Notation}

Let $\mathcal{S} = \{(\mathbf{x}_t, y_t)\}_{t=1}^{\infty}$ denote the data stream, $\mathcal{D} = \{D_1, \ldots, D_K\}$ the detector portfolio, and $\boldsymbol{\phi}_t \in \mathbb{R}^m$ the meta-feature vector at time $t$. For LinUCB, we maintain for each detector $i$: the design matrix $\mathbf{A}_i = \lambda_{\text{reg}}\mathbf{I} + \sum_{s:i_s=i} \boldsymbol{\phi}_s \boldsymbol{\phi}_s^\top$ and the response vector $\mathbf{b}_i = \sum_{s:i_s=i} r_s \boldsymbol{\phi}_s$. The ridge regression estimate is $\hat{\boldsymbol{\theta}}_i = \mathbf{A}_i^{-1} \mathbf{b}_i$.

\subsection{Proof of Theorem 1 (Level 1 Regret)}

\textbf{Theorem 1.} \textit{Under assumptions A1--A3, with exploration parameter $\alpha_{\text{ucb}} = \sigma_{\text{noise}}\sqrt{m \log(KT/\delta_{\text{conf}})} + S$, LinUCB detector selection achieves with probability at least $1 - \delta_{\text{conf}}$:}
\begin{equation}
R_1(T) = \sum_{t=1}^{T} \left( \max_{i} \boldsymbol{\phi}_t^\top \boldsymbol{\theta}_i^* - \boldsymbol{\phi}_t^\top \boldsymbol{\theta}_{i_t}^* \right) = O(\sqrt{mT \log T})
\end{equation}

\textbf{Proof.} The proof proceeds in three steps.

\textit{Step 1: Confidence Ellipsoid Construction.}
By the self-normalized martingale inequality \cite{Abbasi2011Improved}, for any $\delta_{\text{conf}} > 0$, with probability at least $1 - \delta_{\text{conf}}/K$, for all $t \geq 1$:
\begin{equation}
\|\hat{\boldsymbol{\theta}}_{i,t} - \boldsymbol{\theta}_i^*\|_{\mathbf{A}_{i,t}} \leq \sigma_{\text{noise}}\sqrt{2\log\left(\frac{\det(\mathbf{A}_{i,t})^{1/2} \det(\lambda_{\text{reg}}\mathbf{I})^{-1/2}}{\delta_{\text{conf}}/K}\right)} + \sqrt{\lambda_{\text{reg}}}S
\end{equation}
Using the determinant-trace inequality $\det(\mathbf{A}_{i,t}) \leq \left(\frac{\text{tr}(\mathbf{A}_{i,t})}{m}\right)^m \leq \left(\lambda_{\text{reg}} + \frac{tL^2}{m}\right)^m$ and applying a union bound over all $K$ detectors, with probability at least $1 - \delta_{\text{conf}}$, for all $i \in [K]$ and $t \geq 1$:
\begin{equation}
\|\hat{\boldsymbol{\theta}}_{i,t} - \boldsymbol{\theta}_i^*\|_{\mathbf{A}_{i,t}} \leq \alpha_{\text{ucb},t}
\end{equation}
where $\alpha_{\text{ucb},t} = \sigma_{\text{noise}}\sqrt{m\log(1 + tL^2/\lambda_{\text{reg}}) + 2\log(K/\delta_{\text{conf}})} + \sqrt{\lambda_{\text{reg}}}S \leq \alpha_{\text{ucb}}$ for our choice of $\alpha_{\text{ucb}}$.

\textit{Step 2: Instantaneous Regret Bound.}
Let $i^* = \arg\max_i \boldsymbol{\phi}_t^\top \boldsymbol{\theta}_i^*$ be the optimal detector at time $t$. The UCB selection rule chooses:
\begin{equation}
i_t = \arg\max_i \left( \boldsymbol{\phi}_t^\top \hat{\boldsymbol{\theta}}_i + \alpha_{\text{ucb}} \|\boldsymbol{\phi}_t\|_{\mathbf{A}_i^{-1}} \right)
\end{equation}
Under the confidence event, we have:
\begin{align}
\boldsymbol{\phi}_t^\top \boldsymbol{\theta}_{i^*}^* &\leq \boldsymbol{\phi}_t^\top \hat{\boldsymbol{\theta}}_{i^*} + \alpha_{\text{ucb}} \|\boldsymbol{\phi}_t\|_{\mathbf{A}_{i^*}^{-1}} \\
&\leq \boldsymbol{\phi}_t^\top \hat{\boldsymbol{\theta}}_{i_t} + \alpha_{\text{ucb}} \|\boldsymbol{\phi}_t\|_{\mathbf{A}_{i_t}^{-1}} \\
&\leq \boldsymbol{\phi}_t^\top \boldsymbol{\theta}_{i_t}^* + 2\alpha_{\text{ucb}} \|\boldsymbol{\phi}_t\|_{\mathbf{A}_{i_t}^{-1}}
\end{align}
where the first inequality uses the confidence bound, the second uses the UCB selection, and the third applies the confidence bound again. Thus, the instantaneous regret is:
\begin{equation}
r_t^{(1)} = \boldsymbol{\phi}_t^\top \boldsymbol{\theta}_{i^*}^* - \boldsymbol{\phi}_t^\top \boldsymbol{\theta}_{i_t}^* \leq 2\alpha_{\text{ucb}} \|\boldsymbol{\phi}_t\|_{\mathbf{A}_{i_t}^{-1}}
\end{equation}

\textit{Step 3: Cumulative Regret via Elliptical Potential Lemma.}
By the elliptical potential lemma \cite{Dani2008Stochastic}:
\begin{equation}
\sum_{t=1}^T \|\boldsymbol{\phi}_t\|_{\mathbf{A}_{i_t,t-1}^{-1}}^2 \leq 2m\log\left(1 + \frac{TL^2}{\lambda_{\text{reg}} m}\right)
\end{equation}
Applying Cauchy-Schwarz:
\begin{align}
R_1(T) &= \sum_{t=1}^T r_t^{(1)} \leq 2\alpha_{\text{ucb}} \sum_{t=1}^T \|\boldsymbol{\phi}_t\|_{\mathbf{A}_{i_t}^{-1}} \\
&\leq 2\alpha_{\text{ucb}} \sqrt{T \sum_{t=1}^T \|\boldsymbol{\phi}_t\|_{\mathbf{A}_{i_t}^{-1}}^2} \\
&\leq 2\alpha_{\text{ucb}} \sqrt{2mT\log\left(1 + \frac{TL^2}{\lambda_{\text{reg}} m}\right)} \\
&= O(\sqrt{mT \log T})
\end{align}
This completes the proof. \hfill $\square$

\subsection{Proof of Theorem 2 (Level 2 Regret)}

\textbf{Theorem 2.} \textit{For Thompson Sampling on $C$ configurations per detector, with $N_i(T)$ selections of detector $i$, the cumulative regret satisfies:}
\begin{equation}
R_2(T) = \sum_{i=1}^{K} O(\sqrt{C \cdot N_i(T) \log N_i(T)})
\end{equation}

\textbf{Proof.} We analyze the hyperparameter selection problem for each detector separately.

\textit{Step 1: Reduction to Multi-Armed Bandit.}
Conditioned on detector $i$ being selected at times $\{t : i_t = i\}$, the hyperparameter selection problem reduces to a standard $C$-armed bandit over $N_i(T)$ rounds. The rewards $r_t \in [-1, 1]$ are normalized to $\tilde{r}_t = (r_t + 1)/2 \in [0, 1]$.

\textit{Step 2: Beta-Bernoulli Thompson Sampling Analysis.}
For bounded rewards in $[0,1]$, we model the normalized reward as a Bernoulli-like outcome and maintain Beta posterior distributions. By Theorem 1 of \cite{Agrawal2013Further}, for a $C$-armed bandit with bounded rewards, Thompson Sampling with Beta$(1,1)$ priors achieves expected regret:
\begin{equation}
\mathbb{E}[R_i(N_i)] \leq O\left(\sum_{c: \Delta_c > 0} \frac{\log N_i}{\Delta_c}\right)
\end{equation}
where $\Delta_c = \mu^* - \mu_c$ is the suboptimality gap for configuration $c$. This can be bounded as:
\begin{equation}
\mathbb{E}[R_i(N_i)] \leq O(\sqrt{C \cdot N_i(T) \log N_i(T)})
\end{equation}
using the gap-free bound from \cite{Agrawal2012Thompson}.

\textit{Step 3: Aggregation Across Detectors.}
Since the detector selections partition the time horizon (i.e., $\sum_{i=1}^K N_i(T) = T$), and each detector's hyperparameter selection is an independent bandit problem, the total Level 2 regret is:
\begin{equation}
R_2(T) = \sum_{i=1}^{K} R_i(N_i(T)) = \sum_{i=1}^{K} O(\sqrt{C \cdot N_i(T) \log N_i(T)})
\end{equation}
This completes the proof. \hfill $\square$

\subsection{Proof of Corollary (Combined Regret)}

\textbf{Corollary.} \textit{Let $(i^*, c^*)$ denote the optimal detector-configuration pair. The total regret decomposes as $R(T) = R_1(T) + R_2(T)$, where $R_1(T)$ captures suboptimality from detector selection and $R_2(T)$ from configuration selection. The combined bound is:}
\begin{equation}
R(T) = R_1(T) + R_2(T) = O(\sqrt{mT \log T} + \sqrt{CKT \log T})
\end{equation}

\textbf{Proof.} We combine the regret bounds from Theorems 1 and 2.

From Theorem 1, $R_1(T) = O(\sqrt{mT \log T})$.

For $R_2(T)$, we apply the Cauchy-Schwarz inequality:
\begin{align}
R_2(T) &= \sum_{i=1}^{K} O(\sqrt{C \cdot N_i(T) \log N_i(T)}) \\
&\leq O\left(\sqrt{C \log T} \sum_{i=1}^{K} \sqrt{N_i(T)}\right) \\
&\leq O\left(\sqrt{C \log T} \cdot \sqrt{K \sum_{i=1}^{K} N_i(T)}\right) \\
&= O\left(\sqrt{C \log T} \cdot \sqrt{KT}\right) \\
&= O(\sqrt{CKT \log T})
\end{align}
where we used $\log N_i(T) \leq \log T$, Cauchy-Schwarz ($\sum_i \sqrt{a_i} \leq \sqrt{K \sum_i a_i}$), and $\sum_i N_i(T) = T$.

Therefore:
\begin{equation}
R(T) = R_1(T) + R_2(T) = O(\sqrt{mT \log T} + \sqrt{CKT \log T})
\end{equation}

Since both terms grow as $O(\sqrt{T \log T})$, the average regret vanishes:
\begin{equation}
\lim_{T \to \infty} \frac{R(T)}{T} = \lim_{T \to \infty} O\left(\sqrt{\frac{\log T}{T}}\right) = 0
\end{equation}
This guarantees that AutoDD converges to the optimal detector-configuration pair as $T \to \infty$. \hfill $\square$

\subsection{Sample Complexity}

To achieve average regret at most $\epsilon$, we require $R(T)/T \leq \epsilon$. From the combined regret bound:
\begin{equation}
\frac{R(T)}{T} \leq \frac{c_1\sqrt{mT\log T} + c_2\sqrt{CKT\log T}}{T} = \frac{(c_1\sqrt{m} + c_2\sqrt{CK})\sqrt{\log T}}{\sqrt{T}}
\end{equation}
for constants $c_1, c_2$. Setting this equal to $\epsilon$ and solving:
\begin{equation}
T = O\left(\frac{(m + CK)\log(K/\delta_{\text{conf}})}{\epsilon^2}\right)
\end{equation}
With $m = 15$ meta-features, $K = 7$ detectors, and $C = 5$ configurations, achieving $\epsilon = 0.01$ average regret requires approximately $T \approx 5 \times 10^5$ windows, which is reached early in typical streaming applications.

\subsection{Reward Function Properties}

The reward function defined in Eq.~(7) of the main paper is designed to align bandit feedback with drift detection objectives. We establish key properties that justify this design.

\textbf{Proposition 1 (Boundedness).} \textit{The reward function $r_t \in [-1, 1]$ for all $t$.}

\textit{Proof.} By construction: (i) when $d_t=1, \Delta a_t > \tau$: $r_t = 0.5 + \min(\Delta a_t \cdot 2, 0.5) \in [0.5, 1.0]$; (ii) when $d_t=1, \Delta a_t \leq \tau$: $r_t = -0.3$; (iii) when $d_t=0, \Delta a_t > \tau$: $r_t = -0.5 - \min(\Delta a_t, 0.5) \in [-1.0, -0.5]$; (iv) when $d_t=0, \Delta a_t \leq \tau$: $r_t = 0.1$. Thus $r_t \in [-1, 1]$. \hfill $\square$

\textbf{Proposition 2 (Incentive Compatibility).} \textit{The reward function is incentive-compatible: it maximizes expected reward when the detector correctly identifies drift.}

\textit{Proof.} Consider the four scenarios:
\begin{itemize}
\item \textbf{True Positive} ($d_t=1$, drift present with $\Delta a_t > \tau$): Reward is positive ($r_t \geq 0.5$), incentivizing detection.
\item \textbf{False Positive} ($d_t=1$, no drift with $\Delta a_t \leq \tau$): Reward is negative ($r_t = -0.3$), penalizing false alarms.
\item \textbf{False Negative} ($d_t=0$, drift present with $\Delta a_t > \tau$): Reward is strongly negative ($r_t \leq -0.5$), heavily penalizing missed drifts.
\item \textbf{True Negative} ($d_t=0$, no drift with $\Delta a_t \leq \tau$): Reward is small positive ($r_t = 0.1$), mildly rewarding stability.
\end{itemize}
The asymmetric penalties ($-0.5$ to $-1.0$ for false negatives vs. $-0.3$ for false positives) reflect that missed drifts typically cause more severe performance degradation than false alarms. \hfill $\square$

\subsection{Justification of Linear Reward Assumption}

Assumption A2 posits that expected reward is linear in meta-features: $\mathbb{E}[r_{i,t} \mid \boldsymbol{\phi}_t] = \boldsymbol{\phi}_t^\top \boldsymbol{\theta}_i^*$. We provide theoretical and empirical justification.

\textbf{Proposition 3 (Approximate Linearity).} \textit{Under mild smoothness conditions on the detector-stream interaction, the expected reward can be approximated by a linear function with bounded error.}

\textit{Proof Sketch.} Let $f_i: \mathbb{R}^m \to [-1,1]$ denote the true reward function for detector $i$. For meta-features normalized to $[0,1]^m$, if $f_i$ is $L$-Lipschitz continuous, then by standard function approximation theory, there exists $\boldsymbol{\theta}_i$ such that:
\begin{equation}
\sup_{\boldsymbol{\phi} \in [0,1]^m} |f_i(\boldsymbol{\phi}) - \boldsymbol{\phi}^\top \boldsymbol{\theta}_i| \leq \epsilon_{\text{approx}}
\end{equation}
where $\epsilon_{\text{approx}} = O(L/\sqrt{m})$ for appropriately chosen $\boldsymbol{\theta}_i$. The approximation error can be absorbed into the sub-Gaussian noise term in Assumption A3 with adjusted variance $\sigma_{\text{noise}}^2 + \epsilon_{\text{approx}}^2$, preserving the regret bound structure with slightly larger constants. \hfill $\square$

\textbf{Empirical Verification.} We fit linear models $\hat{r}_{i,t} = \boldsymbol{\phi}_t^\top \hat{\boldsymbol{\theta}}_i$ to observed rewards across all experiments. The average $R^2$ coefficient across detectors is 0.73, indicating that linear models explain substantial reward variance. Residual analysis shows approximately symmetric, zero-mean errors consistent with the sub-Gaussian assumption.

\subsection{Meta-Feature Informativeness}

We establish that the designed meta-features provide sufficient information to distinguish detector effectiveness across different stream conditions.

\textbf{Lemma 1 (Feature Discriminability).} \textit{The 15 meta-features partition the stream state space into regions where different detectors are optimal.}

\textit{Proof Sketch.} We show each feature category captures distinct aspects of detector suitability:

\textbf{Statistical features} (velocity, dimensionality, sparsity, noise, imbalance): These determine computational constraints and signal-to-noise ratio. High-velocity streams favor lightweight detectors (ADWIN, Page-Hinkley); high-dimensional sparse data favors distribution-based methods (KSWIN).

\textbf{Stationarity features} (error volatility, KL divergence, autocorrelation, mean/variance shift): These directly measure distribution stability. High KL divergence indicates distribution drift favoring KSWIN; high error volatility without distribution shift suggests virtual drift favoring error-based detectors (DDM, EDDM).

\textbf{Temporal features} (trend, periodicity, change frequency/magnitude, recurrence): These capture drift dynamics. High periodicity suggests recurring patterns favoring detectors with memory (ADWIN); gradual trends favor sensitive detectors (EDDM); sudden changes favor responsive detectors (DDM, HDDM-A).

Formally, let $\mathcal{R}_i = \{\boldsymbol{\phi} : \boldsymbol{\theta}_i^{*\top} \boldsymbol{\phi} \geq \boldsymbol{\theta}_j^{*\top} \boldsymbol{\phi}, \forall j\}$ denote the region where detector $i$ is optimal. The feature design ensures $\bigcup_{i=1}^K \mathcal{R}_i = [0,1]^m$ (coverage) and $\text{vol}(\mathcal{R}_i) > 0$ for all $i$ (non-degeneracy), meaning each detector is optimal in some non-trivial region of the feature space. \hfill $\square$

\subsection{Portfolio Diversity}

\textbf{Proposition 4 (Diversity Benefit).} \textit{A diverse detector portfolio achieves lower worst-case regret than any single detector.}

\textit{Proof.} Let $D^*(\boldsymbol{\phi})$ denote the optimal detector for context $\boldsymbol{\phi}$. For a single fixed detector $D_j$, the worst-case expected regret over contexts is:
\begin{equation}
\max_{\boldsymbol{\phi}} \left( \mathbb{E}[r_{D^*(\boldsymbol{\phi})}] - \mathbb{E}[r_{D_j}] \right) = \Delta_{\max}^{(j)}
\end{equation}
where $\Delta_{\max}^{(j)} > 0$ unless $D_j$ dominates all detectors uniformly (which empirically does not hold).

With a portfolio and LinUCB selection, the regret after $T$ rounds is $O(\sqrt{mT\log T})$, which yields average regret $O(\sqrt{m\log T/T}) \to 0$ as $T \to \infty$. Thus, for sufficiently large $T$:
\begin{equation}
\frac{1}{T}\sum_{t=1}^T r_t^{\text{portfolio}} > \max_j \frac{1}{T}\sum_{t=1}^T r_t^{(j)} - \epsilon
\end{equation}
for arbitrarily small $\epsilon$, demonstrating the portfolio's asymptotic optimality. \hfill $\square$
